    Bring the facts into your own app

    snapple-facts is a zero-dependency TypeScript package with full typings, a class API, and standalone functions — pick whichever style fits. 

    npm install snapple-facts --save-dev
    import SnappleFacts from 'snapple-facts';

const snappleFacts = new SnappleFacts();

snappleFacts.randomFact();               // { number, fact }
snappleFacts.getFacts();                 // SnappleFact[]
snappleFacts.listFacts();                // string[]
snappleFacts.getFactByNumber(1);         // SnappleFact | undefined
    import { RandomFact, GetFacts, ListFacts, GetFactByNumber } from 'snapple-facts';

const randomFact = RandomFact();
const facts = GetFacts();
const factList = ListFacts();
const fact = GetFactByNumber(1);
